As muito próximas do topo são granulares demais, sendo importante definir um tamanho de tarefa mínimo
e um máximo. O artigo do SSCAD ia de um tamanho de bloco de 12 (segmentação 32)
a um bloco de tamanho 96 (segmentação 4). Acho que o tamanho pequeno para o bloco se justifica,
mas para o tamanho máximo acho que podemos usar um maior, tamanho de bloco de 128 (segmentação 3),
que será 27 tarefas por iteração ao todo, gerando menos tarefas do que há threads para todas as máquinas.

#+BEGIN_SRC R :session *R* :async yes
  library(numbers)
  d <- divisors(384)
  d[d >= 12 & d <= 128] |> paste(collapse = ",")

Nos testes, matemos a análise de IO, considerando com e sem.
Para corroborar o IO, é interessante realizar um estudo do thoughput do sistema
 - Segundo o menino, um cálculo base de Volume de dados/tempo é suficiente, desde que
   contenha também métricas do I/O do sistema.
   Para isso, pode-se empregar um benchmarker como /fio/ para verificar o I/O do sistema
   e se basear nisso. Também podemos usar os traces para calcular o tempo de I/O total gasto e o
   overlap, que seria uma métrica reveladora. Para o /fletcher-base/ o tempo de I/O já é computado
   e é simples.

Outro experimento, que usa as mesmas métricas acima é o teste de escalabilidade.
Dado uma configuração, o que acontece ao aumentar o número de trabalhadores StarPU e
o número de OMP threads?

Temos agora que gerar alguns experimentos

*** Experimentos do star-fletcher

Experimento matriz
- Fixa o volume e altera a segmentação.
- Ele serve para caracterizar bem a execução da aplicação. Importante aumentar o número de replicações.
     
Experimento do trace
- Para demonstar o overlap de tarefas e qualificar o desempenho do I/O
  
Experimento de escalabilidade
- Dado uma configuração específica, como que ele performa ao aumentar o número de trabalhadores
- Primeiro aumentar os p-cores e depois ir incrementando os e-cores
     
*** Experimento do /fletcher-base/

Experimento matriz
- Altera as características do OpenMP

Experimento de escalabilidade
- Aumentar o número de OMP_WORKERS, primeiro usando os p-cores e depois os e-cores


#+BEGIN_SRC R :session *R* :async yes :results output silent 
   source("./libExp.R")
   options(scipen = 999)

   total_width = 384
   border_size = 4
   absorb_size = 8
   width = total_width - 2 * (border_size + absorb_size)
   seed = 0
   time_step = 0.0001
   iterations = 20
   replications = 10

   divs = divisors(total_width)
   segmentations = divs[divs >= 12 & divs <= 128] |> (\(x) total_width / x )()

   sf_cpu_factors <- list(
     BlockSeg = segmentations,
     OutputTime = c(0, 1, 5) * time_step
   )

   sf_ex <- generate_experiment(sf_cpu_factors, replications, seed) |>
     as_tibble() |>
     mutate(
       WithIO = if_else(OutputTime == 0, 0, 1),
       OutputTime = if_else(OutputTime == 0, 1, as.numeric(as.character(OutputTime))),
       Width = width,
       AbsorbSize = absorb_size,
       TotalTime = format(time_step * iterations, scientific = FALSE),
       TimeStep = format(time_step, scientific = FALSE),
       Blocks = str_sub(Blocks, 2, 3)
     )
   # for the fletcher-base, remove all non-relevant parameters and add the new ones
  fb_ex <- sf_ex |> select(-BlockSeg) |> unique() |>
    cross_join(tibble(Schedulers = c("static", "dynamic", "guided")))

  sf_ex |> readr::write_csv(glue("./pdp/star-fletcher-exp.csv"))
  fb_ex |> readr::write_csv(glue("./pdp/fletcher-base-exp.csv"))
#+END_SRC

  msamples_plot <- function(sft, fbt) {
    x_breaks <- sort(unique(sft$Order))
    
    fb_stats <- fbt |>
      mutate(Machine = str_sub(Machine, end = -2)) |>
      filter(WithIO == 0) |>
      group_by(Machine) |>
      summarize(
        rep = n(),
        FB_Mean   = mean(Msamples),
        FB_Sd     = sd(Msamples),
        FB_Error  = (3 * FB_Sd) / sqrt(n()),
        .groups   = "drop" 
      )

    sf_msamples_plot <- sft |>
      select(-FILE) |>
      filter(WithIO == 0) |>
      mutate(Machine = str_sub(Machine, end = -2)) |>
      group_by(across(-c(Msamples, ExecTime, Blocks))) |>
      summarize(
        rep = n(),
        Msamples_mean = mean(Msamples),
        Msamples_sd = sd(Msamples),
        Msamples_error = (3 * Msamples_sd) / sqrt(rep),
        
        ExecTime_mean = mean(ExecTime),
        ExecTime_sd = sd(ExecTime),
        ExecTime_error = (3 * ExecTime_sd) / sqrt(rep),
        .groups = "drop"
      ) 

    ggplot(sf_msamples_plot, aes(x = Order, y = Msamples_mean, color = Machine)) +
      
      geom_point(alpha = 0.8) +
      geom_errorbar(aes(ymin = Msamples_mean - Msamples_error, ymax = Msamples_mean + Msamples_error, color = Machine), width = 0.5) +
      # Optional line connecting groups
      # geom_line(linewidth = 0.8) +
      
      geom_rect(
  	data = fb_stats, 
  	aes(ymin = FB_Mean - FB_Error, ymax = FB_Mean + FB_Error, fill = Machine),
	  xmin = -Inf,
	  xmax = Inf,
	  inherit.aes = FALSE,

  time_plot <- function(sft, fbt) {
    x_breaks <- c(12, 32, 48, 64, 96)
    
    fb_stats <- fbt |>
      select(-FILE) |>
      mutate(Machine = str_sub(Machine, end = -2)) |>
      group_by(across(-c(Msamples, ExecTime, Blocks))) |>
      summarize(
        fb_n = n(),
        ExecTime_fb_mean = mean(ExecTime, na.rm = TRUE),
        ExecTime_fb_sd = sd(ExecTime, na.rm = TRUE),
        .groups = "drop"
      )

    sf_comb <- sft |>
      select(-FILE) |>
      mutate(Machine = str_sub(Machine, end = -2)) |>
      group_by(across(-c(Msamples, ExecTime, Blocks))) |>
      summarize(
        sf_n = n(),
        ExecTime_sf_mean = mean(ExecTime, na.rm = TRUE),
        ExecTime_sf_sd = sd(ExecTime, na.rm = TRUE),
        .groups = "drop"
      ) |>
      left_join(fb_stats, by = c("Machine", "WithIO")) |>
      mutate(
        # Difference of Means
        Diff_mean = ExecTime_sf_mean - ExecTime_fb_mean,
        
        # Combined Standard Error for Independent Samples
        Combined_SE = sqrt((ExecTime_sf_sd^2 / sf_n) + (ExecTime_fb_sd^2 / fb_n)),
        
        # 3x Error mapping
        Diff_error = 3 * Combined_SE
      )

    # 3. Plot (Exact same plotting code as above)
    ggplot(sf_comb, aes(x = Order, y = Diff_mean, color = factor(WithIO))) +
      geom_hline(yintercept = 0, linetype = "dashed") +
      geom_point(alpha = 0.8) +
      geom_line(linewidth = 0.8) +
      geom_errorbar(
        aes(ymin = Diff_mean - Diff_error, ymax = Diff_mean + Diff_error),
