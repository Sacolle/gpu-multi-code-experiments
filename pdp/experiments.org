Experimento de escalabilidade do /star-fletcher/ e do /fletcher-base/.
Ele é feito por máquina, pois depende da topologia de cada CPU (ver [[file:../cpu-topology.org][cpu-topology.org]]).
A cada execução aumenta-se o número de CPUs usados, sempre com um thread por core físico:
primeiro os p-cores e depois os e-cores, e nas máquinas com dois sockets enche-se um socket antes do outro.
Por último, nas máquinas com SMT, há um caso com todos os threads (hyperthreading).
O número de CPUs segue potências de 2 mais os pontos de transição (socket inteiro, todos os p-cores, todos os cores físicos).
- O /star-fletcher/ usa a segmentação com melhor desempenho sem IO de cada máquina no experimento matriz.
- O /fletcher-base/ usa o escalonador =dynamic=, o melhor em todas as máquinas.
- Os dois rodam sem IO e com IO a cada passo.
- Os dois são fixados nos mesmos CPUs: =CpuList= (ids do Linux) vai para o =OMP_PLACES= e
  =StarpuCpuList= (índices lógicos do hwloc, que é o que o StarPU espera) para o =STARPU_WORKERS_CPUID=.

#+BEGIN_SRC R :session *R* :async yes :results output silent
  source("./libExp.R")
  options(scipen = 999)

  # Ordem em que os CPUs de cada máquina são adicionados (ids do Linux, ver cpu-topology.org):
  # primeiro um thread por core físico (p-cores antes dos e-cores, enchendo um socket antes do
  # outro) e, por último, um caso com todos os threads (hyperthreading) nas máquinas com SMT.
  # O StarPU usa o índice lógico do hwloc em STARPU_WORKERS_CPUID, que é diferente do id do Linux
  # nas máquinas com SMT, por isso `logical` faz a conversão.
  smt_logical <- \(cores) \(p) if_else(p < cores, 2 * p, 2 * (p - cores) + 1)

  scal_machines <- list(
    cei = list(
      block_seg = 8,
      physical  = 0:23,
      smt       = 0:47,
      counts    = c(1, 2, 4, 8, 12, 16, 24),  # 12: um socket inteiro
      logical   = smt_logical(24)
    ),
    draco = list(
      block_seg = 16,
      physical  = c(seq(0, 14, 2), seq(1, 15, 2)),  # ids pares no socket 0, ímpares no socket 1
      smt       = 0:31,
      counts    = c(1, 2, 4, 8, 16),  # 8: um socket inteiro
      logical   = \(p) {
        q <- p %% 16
        core <- if_else(q %% 2 == 0, q / 2, 8 + (q - 1) / 2)
        2 * core + (p >= 16)
      }
    ),
    grace = list(
      block_seg = 8,
      physical  = 0:143,
      smt       = NULL,
      counts    = c(1, 2, 4, 8, 16, 32, 64, 72, 128, 144),  # 72: um socket inteiro
      logical   = identity
    ),
    lunaris = list(
      block_seg = 8,
      physical  = 0:23,
      smt       = 0:47,
      counts    = c(1, 2, 4, 8, 16, 24),
      logical   = smt_logical(24)
    ),
    poti = list(
      block_seg = 8,
      physical  = c(seq(0, 14, 2), 16:27),  # 8 p-cores, depois 12 e-cores
      smt       = 0:27,
      counts    = c(1, 2, 4, 8, 16, 20),    # 8: todos os p-cores
      logical   = identity
    ),
    tupi = list(
      block_seg = 16,
      physical  = c(seq(0, 14, 2), 16:31),  # 8 p-cores, depois 16 e-cores
      smt       = 0:31,
      counts    = c(1, 2, 4, 8, 16, 24),    # 8: todos os p-cores
      logical   = identity
    )
  )

  cpu_points <- function(m) {
    sets <- c(map(m$counts, \(n) m$physical[1:n]), if (is.null(m$smt)) list() else list(m$smt))
    tibble(
      NumCPUs       = lengths(sets),
      SMT           = c(rep(0, length(m$counts)), rep(1, length(sets) - length(m$counts))),
      CpuList       = map_chr(sets, \(s) paste(s, collapse = " ")),
      StarpuCpuList = map_chr(sets, \(s) paste(m$logical(s), collapse = " "))
    )
  }

  dir.create("./pdp/scalability", showWarnings = FALSE)

  iwalk(scal_machines, \(m, machine) {
    points <- cpu_points(m)

    sf_scal <- generate_experiment(list(NumCPUs = points$NumCPUs, OutputTime = c(1, time_step)), 5, seed) |>
      as_tibble() |>
      mutate(
        NumCPUs = as.numeric(as.character(NumCPUs)),
        OutputTime = as.numeric(as.character(OutputTime)),
        WithIO = if_else(OutputTime == 1, 0, 1),  # OutputTime 1 > TotalTime: nunca escreve
        OutputTime = format(OutputTime, scientific = FALSE),
        BlockSeg = m$block_seg,
        Width = width,
        AbsorbSize = absorb_size,
        TotalTime = format(time_step * iterations, scientific = FALSE),
        TimeStep = format(time_step, scientific = FALSE),
        Blocks = str_sub(Blocks, 2, 3)
      ) |>
      left_join(points, by = "NumCPUs")

    fb_scal <- sf_scal |>
      select(-BlockSeg, -StarpuCpuList) |>
      cross_join(tibble(Schedulers = c("static", "dynamic", "guided")))

    sf_scal |> readr::write_csv(glue("./pdp/scalability/star-fletcher-scal-{machine}.csv"))
    fb_scal |> readr::write_csv(glue("./pdp/scalability/fletcher-base-scal-{machine}.csv"))
  })

  time_plot <- function(sft, fbt, name = "") {
    x_breaks <- sort(unique(sft$Order))

    # readable label for the output frequency, based on WithIO so it works
    # whether the runs without IO have OutputTime 0 or 1
    io_label <- function(df) {
      df |>
        mutate(
          Steps = round(as.numeric(as.character(OutputTime)) / as.numeric(as.character(TimeStep))),
          IO = if_else(WithIO == 0, "Without IO", paste0("Output every ", Steps, " steps")),
          IO = fct_relevel(IO, "Without IO", after = Inf)
        )
    }

    fb_stats <- fbt |>
      mutate(Machine = sub("\\d+$", "", Machine)) |>
      io_label() |>
      group_by(Machine, IO, Schedulers) |>
      summarize(
        fb_n = n(),
        ExecTime_fb_mean = mean(ExecTime, na.rm = TRUE),
        ExecTime_fb_sd = sd(ExecTime, na.rm = TRUE),
        .groups = "drop"
      )

    sf_comb <- sft |>
      mutate(Machine = sub("\\d+$", "", Machine)) |>
      io_label() |>
      group_by(Machine, IO, Order) |>
      summarize(
        sf_n = n(),
        ExecTime_sf_mean = mean(ExecTime, na.rm = TRUE),
        ExecTime_sf_sd = sd(ExecTime, na.rm = TRUE),
        .groups = "drop"
      ) |>
      # each star-fletcher point is compared with each of the 3 OpenMP schedulers
      # that ran with the same output frequency
      inner_join(fb_stats, by = c("Machine", "IO"), relationship = "many-to-many") |>
      mutate(
        # Difference of Means
        Diff_mean = ExecTime_sf_mean - ExecTime_fb_mean,

        # Combined Standard Error for Independent Samples
        Combined_SE = sqrt((ExecTime_sf_sd^2 / sf_n) + (ExecTime_fb_sd^2 / fb_n)),

        # 3x Error mapping
        Diff_error = 3 * Combined_SE
      )

    ggplot(sf_comb, aes(x = Order, y = Diff_mean, color = Schedulers)) +
      geom_hline(yintercept = 0, linetype = "dashed") +
      geom_point(alpha = 0.8) +
      geom_line(linewidth = 0.8) +
      geom_errorbar(
        aes(ymin = Diff_mean - Diff_error, ymax = Diff_mean + Diff_error), 
        width = 0.5
      ) +
      scale_x_continuous(breaks = x_breaks) +
      facet_grid(Machine ~ IO, scales = "free_y") +
      labs(
        title = paste0(name, "Execution Time Difference vs. Block Order"),
        subtitle = "Below 0: star-fletcher is faster. Rows: machine, columns: output frequency",
        x = "Block Size",
        y = "Execution Time Difference (s)",
        color = "OpenMP scheduler"
      ) +
      theme_bw(base_size=15) +
      theme(
        legend.position = "bottom",
        legend.margin = margin(t = -10, b = -10),
        axis.text.x = element_text(size = 10)
      )
  }

** Análise dos traces

Os traces foram organizados por máquina, OutputTime e BlockSeg e convertidos com a fase 1 do StarVZ
pelo script =organize-traces.sh=. Estatísticas de um trace do tupi com ordem 24 (BlockSeg 16) e IO a cada passo:
duração das tarefas, propagação nos p-cores e e-cores, ocupação dos workers, tempo do runtime e
sobreposição do IO com a computação.

#+BEGIN_SRC R :session *R* :async yes :results output
  library(starvz)
  options(width = 200)

  # ordem 24 = BlockSeg 16 (384 / 24), com IO a cada passo
  trace_dir <- "./pdp/results/star-fletcher-cpu-pdp-trace-tree/tupi6/OutputTime-0.0001/BlockSeg-16"
  trace <- starvz_read(trace_dir, selective = FALSE)

  # Os tempos do StarVZ estão em ms. No tupi o worker k fica no CPU k (ver o stdout do StarPU):
  # CPUs 0-15 são os p-cores (2 threads por core) e 16-31 os e-cores.
  tasks <- trace$Application |>
    mutate(
      Worker = as.integer(str_remove(ResourceId, "CPU")),
      CoreKind = if_else(Worker < 16, "P-core", "E-core")
    )
  makespan <- max(tasks$End) - min(tasks$Start)
  n_workers <- n_distinct(tasks$Worker)

  # 1. Duração de cada tipo de tarefa
  task_stats <- tasks |>
    group_by(Task = Value) |>
    summarize(
      n         = n(),
      mean_ms   = mean(Duration),
      sd_ms     = sd(Duration),
      median_ms = median(Duration),
      p95_ms    = quantile(Duration, 0.95),
      total_s   = sum(Duration) / 1000,
      .groups   = "drop"
    ) |>
    mutate(share_pct = 100 * total_s / sum(total_s))

  # 2. A mesma tarefa de propagação nos p-cores e nos e-cores
  kind_stats <- tasks |>
    filter(Value == "wave_propagation") |>
    group_by(CoreKind) |>
    summarize(
      workers   = n_distinct(Worker),
      n         = n(),
      tasks_per_worker = n / workers,
      mean_ms   = mean(Duration),
      sd_ms     = sd(Duration),
      median_ms = median(Duration),
      .groups   = "drop"
    )

  # 3. Ocupação de cada worker: tempo executando tarefas / makespan
  worker_stats <- tasks |>
    group_by(Worker, CoreKind) |>
    summarize(
      n         = n(),
      compute_s = sum(Duration[Value == "wave_propagation"]) / 1000,
      write_s   = sum(Duration[Value == "write_block"]) / 1000,
      busy_s    = sum(Duration) / 1000,
      .groups   = "drop"
    ) |>
    mutate(util_pct = 100 * busy_s / (makespan / 1000))

  util_by_kind <- worker_stats |>
    group_by(CoreKind) |>
    summarize(across(c(n, compute_s, write_s, util_pct), list(mean = mean, min = min, max = max)), .groups = "drop")

  # 4. Tempo gasto pelo runtime (estados do StarPU nos workers)
  runtime_stats <- trace$Starpu |>
    group_by(State = Value) |>
    summarize(n = n(), total_s = sum(Duration) / 1000, mean_us = 1000 * mean(Duration), .groups = "drop") |>
    mutate(pct_worker_time = 100 * total_s / (n_workers * makespan / 1000))

  # 5. Sobreposição do IO com a computação: a linha do tempo é amostrada a cada 1 ms e em cada
  # instante conta-se quantos workers estão escrevendo e quantos estão propagando
  grid <- seq(min(tasks$Start), max(tasks$End), by = 1)
  active <- function(task) {
    t <- filter(tasks, Value == task)
    findInterval(grid, sort(t$Start)) - findInterval(grid, sort(t$End))
  }
  timeline <- tibble(t = grid, writing = active("write_block"), computing = active("wave_propagation"))

  io_overlap <- timeline |>
    summarize(
      pct_time_writing           = 100 * mean(writing > 0),
      pct_time_only_writing      = 100 * mean(writing > 0 & computing == 0),
      workers_computing          = mean(computing),
      workers_computing_if_write = mean(computing[writing > 0])
    )

  cat("Makespan:", round(makespan / 1000, 3), "s com", n_workers, "workers\n\n")
  print(task_stats)
  print(kind_stats)
  print(util_by_kind)
  print(runtime_stats)
  print(io_overlap)
