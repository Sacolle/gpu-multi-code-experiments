As muito próximas do topo são granulares demais, sendo importante definir um tamanho de tarefa mínimo
e um máximo. O artigo do SSCAD ia de um tamanho de bloco de 12 (segmentação 32)
a um bloco de tamanho 96 (segmentação 4). Acho que o tamanho pequeno para o bloco se justifica,
mas para o tamanho máximo acho que podemos usar um maior, tamanho de bloco de 128 (segmentação 3),
que será 27 tarefas por iteração ao todo, gerando menos tarefas do que há threads para todas as máquinas.

*** Cógido de geração dos experimentos de escalabilidade

Experimento de escalabilidade do /star-fletcher/ e do /fletcher-base/.
Ele é feito por máquina, pois depende da topologia de cada CPU (ver [[file:../cpu-topology.org][cpu-topology.org]]).
A cada execução aumenta-se o número de CPUs usados, sempre com um thread por core físico:
primeiro os p-cores e depois os e-cores, e nas máquinas com dois sockets enche-se um socket antes do outro.
Por último, nas máquinas com SMT, há um caso com todos os threads (hyperthreading).
O número de CPUs segue potências de 2 mais os pontos de transição (socket inteiro, todos os p-cores, todos os cores físicos).
- O /star-fletcher/ usa a segmentação com melhor desempenho sem IO de cada máquina no experimento matriz.
- O /fletcher-base/ usa o escalonador =dynamic=, o melhor em todas as máquinas.
- Os dois rodam sem IO e com IO a cada passo.
- Os dois são fixados nos mesmos CPUs: =CpuList= (ids do Linux) vai para o =OMP_PLACES= e
  =StarpuCpuList= (índices lógicos do hwloc, que é o que o StarPU espera) para o =STARPU_WORKERS_CPUID=.

#+BEGIN_SRC R :session *R* :async yes :results output silent
  source("./libExp.R")
  options(scipen = 999)

  # Ordem em que os CPUs de cada máquina são adicionados (ids do Linux, ver cpu-topology.org):
  # primeiro um thread por core físico (p-cores antes dos e-cores, enchendo um socket antes do
  # outro) e, por último, um caso com todos os threads (hyperthreading) nas máquinas com SMT.
  # O StarPU usa o índice lógico do hwloc em STARPU_WORKERS_CPUID, que é diferente do id do Linux
  # nas máquinas com SMT, por isso `logical` faz a conversão.
  smt_logical <- \(cores) \(p) if_else(p < cores, 2 * p, 2 * (p - cores) + 1)

  scal_machines <- list(
    cei = list(
      block_seg = 8,
      physical  = 0:23,
      smt       = 0:47,
      counts    = c(1, 2, 4, 8, 12, 16, 24),  # 12: um socket inteiro
      logical   = smt_logical(24)
    ),
    draco = list(
      block_seg = 16,
      physical  = c(seq(0, 14, 2), seq(1, 15, 2)),  # ids pares no socket 0, ímpares no socket 1
      smt       = 0:31,
      counts    = c(1, 2, 4, 8, 16),  # 8: um socket inteiro
      logical   = \(p) {
        q <- p %% 16
        core <- if_else(q %% 2 == 0, q / 2, 8 + (q - 1) / 2)
        2 * core + (p >= 16)
      }
    ),
    grace = list(
      block_seg = 8,
      physical  = 0:143,
      smt       = NULL,
      counts    = c(1, 2, 4, 8, 16, 32, 64, 72, 128, 144),  # 72: um socket inteiro
      logical   = identity
    ),
    lunaris = list(
      block_seg = 8,
      physical  = 0:23,
      smt       = 0:47,
      counts    = c(1, 2, 4, 8, 16, 24),
      logical   = smt_logical(24)
    ),
    poti = list(
      block_seg = 8,
      physical  = c(seq(0, 14, 2), 16:27),  # 8 p-cores, depois 12 e-cores
      smt       = 0:27,
      counts    = c(1, 2, 4, 8, 16, 20),    # 8: todos os p-cores
      logical   = identity
    ),
    tupi = list(
      block_seg = 16,
      physical  = c(seq(0, 14, 2), 16:31),  # 8 p-cores, depois 16 e-cores
      smt       = 0:31,
      counts    = c(1, 2, 4, 8, 16, 24),    # 8: todos os p-cores
      logical   = identity
    )
  )

  cpu_points <- function(m) {
    sets <- c(map(m$counts, \(n) m$physical[1:n]), if (is.null(m$smt)) list() else list(m$smt))
    tibble(
      NumCPUs       = lengths(sets),
      SMT           = c(rep(0, length(m$counts)), rep(1, length(sets) - length(m$counts))),
      CpuList       = map_chr(sets, \(s) paste(s, collapse = " ")),
      StarpuCpuList = map_chr(sets, \(s) paste(m$logical(s), collapse = " "))
    )
  }

  dir.create("./pdp/scalability", showWarnings = FALSE)

  iwalk(scal_machines, \(m, machine) {
    points <- cpu_points(m)

    sf_scal <- generate_experiment(list(NumCPUs = points$NumCPUs, OutputTime = c(1, time_step)), 5, seed) |>
      as_tibble() |>
      mutate(
        NumCPUs = as.numeric(as.character(NumCPUs)),
        OutputTime = as.numeric(as.character(OutputTime)),
        WithIO = if_else(OutputTime == 1, 0, 1),  # OutputTime 1 > TotalTime: nunca escreve
        OutputTime = format(OutputTime, scientific = FALSE),
        BlockSeg = m$block_seg,
        Width = width,
        AbsorbSize = absorb_size,
        TotalTime = format(time_step * iterations, scientific = FALSE),
        TimeStep = format(time_step, scientific = FALSE),
        Blocks = str_sub(Blocks, 2, 3)
      ) |>
      left_join(points, by = "NumCPUs")

    fb_scal <- sf_scal |>
      select(-BlockSeg, -StarpuCpuList) |>
      cross_join(tibble(Schedulers = c("static", "dynamic")))

    sf_scal |> readr::write_csv(glue("./pdp/scalability/star-fletcher-scal-{machine}.csv"))
    fb_scal |> readr::write_csv(glue("./pdp/scalability/fletcher-base-scal-{machine}.csv"))
  })

  msamples_plot <- function(sft, fbt) {
    x_breaks <- sort(unique(sft$Order))
    
    fb_stats <- fbt |>
      mutate(Machine = sub("\\d+$", "", Machine)) |>
      filter(WithIO == 0, Schedulers != "guided") |>
      select(-FILE) |>
      group_by(Machine, Schedulers, Variant) |>
      summarize(
        rep = n(),
        FB_Mean   = mean(Msamples),
        FB_Sd     = sd(Msamples),
        FB_Error  = (3 * FB_Sd) / sqrt(n()),
        .groups   = "drop" 
      ) |>
      mutate(Version = glue("{Schedulers} ({Variant} threads)"))

    sf_msamples_plot <- sft |>
      select(-FILE) |>
      filter(WithIO == 0) |>
      mutate(Machine = sub("\\d+$", "", Machine)) |>
      group_by(across(-c(Msamples, ExecTime, Blocks))) |>
      summarize(
        rep = n(),
        Msamples_mean = mean(Msamples),
        Msamples_sd = sd(Msamples),
        Msamples_error = (3 * Msamples_sd) / sqrt(rep),
        
        ExecTime_mean = mean(ExecTime),
        ExecTime_sd = sd(ExecTime),
        ExecTime_error = (3 * ExecTime_sd) / sqrt(rep),
        .groups = "drop"
      ) 

    ggplot(sf_msamples_plot, aes(x = Order, y = Msamples_mean)) +
      geom_point(alpha = 0.8) +
      geom_errorbar(aes(ymin = Msamples_mean - Msamples_error, ymax = Msamples_mean + Msamples_error), width = 0.5) +

  # Lê a tabela de tarefas (application.parquet) de todos os traces já convertidos pelo StarVZ.

Uma linha por execução: makespan, ocupação, sobreposição do IO e =busy_last25=, a média de workers
propagando no último quarto do makespan. Quando ela fica bem abaixo de =busy_mean=, faltam tarefas
no fim da execução, como no grace: com 512 tarefas por iteração para 144 workers (3,6 por worker),
as últimas iterações não têm outras iterações com que se sobrepor e rodam uma depois da outra.

#+BEGIN_SRC R :session *R* :async yes :results value :colnames yes
  # número de tarefas em execução em cada instante de `grid` (a linha do tempo amostrada a cada 1 ms)
  active_count <- function(start, end, grid) findInterval(grid, sort(start)) - findInterval(grid, sort(end))

  # Uma linha por execução. busy_last25 é a média de workers propagando no último quarto do
  # makespan: quando ela fica bem abaixo de busy_mean, faltam tarefas no fim da execução.
  trace_overview <- all_tasks |>
    group_by(Machine, Order, BlockSeg, TasksPerIter, Workers, IO) |>
    group_modify(\(t, key) {
      makespan <- max(t$End) - min(t$Start)
      grid <- seq(min(t$Start), max(t$End), by = 1)
      prop <- filter(t, Value == "wave_propagation")
      wr <- filter(t, Value == "write_block")
      computing <- active_count(prop$Start, prop$End, grid)
      writing <- active_count(wr$Start, wr$End, grid)
      last25 <- grid >= min(t$Start) + 0.75 * makespan
      tibble(
        makespan_s            = makespan / 1000,
        # duração das tarefas de propagação (as de escrita estão na tabela de duração abaixo)
        task_min_ms           = min(prop$Duration),
        task_mean_ms          = mean(prop$Duration),
        task_max_ms           = max(prop$Duration),
        util_pct              = 100 * sum(t$Duration) / (key$Workers * makespan),
        write_share_pct       = 100 * sum(wr$Duration) / sum(t$Duration),
        busy_mean             = mean(computing),
        busy_last25           = mean(computing[last25]),
        pct_time_writing      = if (nrow(wr) > 0) 100 * mean(writing > 0) else NA,
        pct_time_only_writing = if (nrow(wr) > 0) 100 * mean(writing > 0 & computing == 0) else NA
      )
    }) |>
    ungroup() |>
    mutate(TasksPerWorker = TasksPerIter / Workers, .after = Workers) |>
    arrange(Machine, IO, BlockSeg) |>
    mutate(across(where(is.double), \(x) round(x, 2)))

  trace_overview |> filter(Machine == "grace" & BlockSeg == 8)

#+BEGIN_SRC R :session *R* :async yes :results output silent
  # Cada execução do star-fletcher foi convertida com a fase 1 do StarVZ logo após rodar e
  # compactada em <NumCPUs>-<BlockSeg>-<Steps>.tar.gz. O arquivo inteiro é extraído ao lado dele
  # (só na primeira vez), para que qualquer execução possa ser aberta com starvz_read e usada
  # com os painéis do StarVZ (panel_st, panel_st_runtime, panel_kiteration, ...).
  sf_scale_trace_root <- glue("{base_dir}/star-fletcher-cpu-pdp-exp-scale-trace")

  sf_scale_trace_dirs <- list.files(sf_scale_trace_root, pattern = "\\.tar\\.gz$", recursive = TRUE, full.names = TRUE) |>
    map(\(f) {
      dir <- str_remove(f, "\\.tar\\.gz$")
      if (!dir.exists(dir)) {
        utils::untar(f, exdir = dirname(f))
      }
      parts <- str_split_1(basename(dir), "-")  # <NumCPUs>-<BlockSeg>-<Steps>
      tibble(
        Machine = sub("\\d+$", "", basename(dirname(dir))),
        NumCPUs = as.integer(parts[1]),
        BlockSeg = as.integer(parts[2]),
        Steps = as.integer(parts[3]),
        Dir = dir
      )
    }) |>
    list_rbind() |>
    mutate(IO = factor(if_else(Steps == 1, "IO every step", "Without IO"), levels = c("Without IO", "IO every step")))

  sf_scale_trace_tasks <- sf_scale_trace_dirs |>
    pmap(\(Machine, NumCPUs, BlockSeg, Steps, Dir, IO) {
      arrow::read_parquet(

      Order = 384 / BlockSeg,

    mutate(Workers = n_distinct(ResourceId)) |>
    ungroup()

#+BEGIN_SRC R :session *R* :async yes :results output silent
  # O fletcher-base com trace imprime o início e o fim da computação e do IO de cada iteração
  # ("Iter N, comp_start: <segundos>"). Uma linha por fase de cada iteração, com os tempos em ms
  # a partir do primeiro comp_start da execução, para comparar com os traces do StarVZ.
  fb_scale_trace_root <- glue("{base_dir}/fletcher-base-pdp-exp-scale-trace")

  fb_scale_trace_phases <- list.files(fb_scale_trace_root, pattern = "^stdout-.*\\.out$", recursive = TRUE, full.names = TRUE) |>
    map(\(f) {
      parts <- str_match(basename(f), "^stdout-(\\d+)-([a-z]+)-(\\d+)\\.out$")  # <NumCPUs>-<Schedulers>-<Steps>
      read_lines(f) |>
        str_match("^Iter (\\d+), (comp|io)_(start|end): ([0-9.]+)$") |>
        as_tibble(.name_repair = ~ c("line", "Iteration", "Phase", "Edge", "Time")) |>
        filter(!is.na(line)) |>
        select(-line) |>
        mutate(Iteration = as.integer(Iteration), Time = as.numeric(Time)) |>
        pivot_wider(names_from = Edge, values_from = Time) |>
        mutate(
          Machine = sub("\\d+$", "", basename(dirname(f))),
          NumCPUs = as.integer(parts[2]),
          Schedulers = parts[3],
          Steps = as.integer(parts[4])
        )
    }) |>
    list_rbind() |>
    group_by(Machine, NumCPUs, Schedulers, Steps) |>
    mutate(
      Start = 1000 * (start - min(start[Phase == "comp"])),
      End = 1000 * (end - min(start[Phase == "comp"])),
      Duration = End - Start
    ) |>
    ungroup() |>
    mutate(IO = factor(if_else(Steps == 1, "IO every step", "Without IO"), levels = c("Without IO", "IO every step"))) |>
    select(Machine, NumCPUs, Schedulers, IO, Iteration, Phase, Start, End, Duration) |>
    arrange(Machine, NumCPUs, Schedulers, IO, Iteration, Phase)

  fb_scale_trace_phases |> count(Machine, NumCPUs, Schedulers, IO, Phase) |> print(n = 70)

#+BEGIN_SRC R :session *R* :async yes :results output silent
  # Painel de iterações (panel_kiteration do StarVZ) do star-fletcher e do fletcher-base, um acima do
  # outro e com o mesmo eixo de tempo. Recebe o trace do star-fletcher já lido com starvz_read (sf) e as
  # fases de uma execução do fletcher-base (linhas de fb_scale_trace_phases). O fletcher-base não tem
  # trace do StarPU, então os seus tempos são colocados no formato que o panel_kiteration espera: cada
  # fase vira uma tarefa na linha da sua iteração. Usa a configuração do trace do star-fletcher.
  kiteration_compare <- function(sf, fb_phases, title){
    # as tarefas passam a se chamar Compute e IO (as de inserção da fonte são ignoradas),
    # com as mesmas cores dos gráficos de barras
    sf$Application <- sf$Application |>
      filter(Value != "wave_insertion") |>
      mutate(Value = if_else(Value == "write_block", "IO", "Compute"))
    sf$Colors <- tibble(Value = c("Compute", "IO"), Color = c("#377eb8", "#e41a1c"))

    fb <- sf
    fb$Application <- fb_phases |>
      transmute(
        Iteration, Subiteration = -1L, Node = factor(0), ResourceId = "CPU0",
        Value = if_else(Phase == "io", "IO", "Compute"),
        Start, End, Duration
      )

    x_end <- max(max(sf$Application$End), max(fb$Application$End))

    # tempo médio de uma iteração: do início da primeira tarefa ao fim da última de cada iteração
    # (no star-fletcher as iterações se sobrepõem, então a soma delas é maior que o makespan)
    bind_rows(
      sf$Application |> mutate(Program = "star-fletcher"),
      fb$Application |> mutate(Program = "fletcher-base (dynamic)")
    ) |>
      filter(Iteration >= 1) |>
      group_by(Program, Iteration) |>
      summarize(iteration_ms = max(End) - min(Start), .groups = "drop_last") |>
      summarize(iterations = n(), mean_iteration_ms = mean(iteration_ms), sd_iteration_ms = sd(iteration_ms)) |>
      print()

    # o nome do programa fica no título do eixo Y (na vertical, à esquerda) e os dois painéis
    # compartilham o eixo X, que só aparece embaixo
    (panel_kiteration(sf, x_start = 0, x_end = x_end, sub_ite = FALSE) + labs(y = "star-fletcher") +
       theme(plot.margin = margin(b = 10))) /
      (panel_kiteration(fb, x_start = 0, x_end = x_end, sub_ite = FALSE) + labs(y = "fletcher-base (dynamic)") + guides(fill = "none")) +
      plot_layout(guides = "collect", axes = "collect") +
      plot_annotation(title = title, subtitle = "Iteration (Y axis) over time") &
      theme(legend.position = "bottom")
  }

*** Fome dos workers

Quanto tempo os workers do /star-fletcher/ ficam sem tarefas (/starving/) e se a submissão de tarefas
consegue ficar à frente da execução. Para cada execução: o tempo total, o tempo gasto submetendo as tarefas
de propagação, o custo de submissão por tarefa, a porcentagem do tempo dos workers sem tarefa, a porcentagem
de tarefas que começam logo após serem submetidas (menos de 1 ms) e a média de tarefas submetidas à frente da
execução. Primeiro para os traces com várias segmentações, em todas as máquinas.

#+BEGIN_SRC R :session *R* :async yes :results value :colnames yes
  # Estatísticas de fome de uma execução (pasta convertida pelo StarVZ), tempos em ms:
  # - starving_pct: fração do tempo dos workers sem nenhuma tarefa
  # - submitting_s: tempo entre a submissão da primeira e da última tarefa de propagação
  # - started_on_submit_pct: tarefas que começaram menos de 1 ms depois de serem submetidas
  # - tasks_ahead: média de tarefas submetidas e ainda não executadas (contador "Submitted" do StarPU)
  starvation_stats <- function(dir) {
    app <- arrow::read_parquet(file.path(dir, "application.parquet"), col_select = c(ResourceId, Start, End, Duration))
    tasks <- arrow::read_parquet(file.path(dir, "tasks.parquet"), col_select = c(Name, SubmitTime, StartTime)) |>
      filter(Name == "wave_propagation")
    workers <- nrow(arrow::read_parquet(file.path(dir, "y.parquet")))  # todos os workers, mesmo os sem tarefas
    t0 <- min(app$Start)
    t1 <- max(app$End)
    submitted <- arrow::read_parquet(file.path(dir, "variable.parquet")) |>
      filter(Type == "Submitted", End > t0, Start < t1) |>
      mutate(Duration = pmin(End, t1) - pmax(Start, t0))

    tibble(
      Workers = workers,
      total_s = (t1 - t0) / 1000,
      submitting_s = (max(tasks$SubmitTime) - min(tasks$SubmitTime)) / 1000,
      us_per_submit = 1000 * (max(tasks$SubmitTime) - min(tasks$SubmitTime)) / nrow(tasks),
      starving_pct = 100 * (1 - sum(app$Duration) / (workers * (t1 - t0))),
      started_on_submit_pct = 100 * mean(tasks$StartTime - tasks$SubmitTime < 1),
      tasks_ahead = weighted.mean(submitted$Value, submitted$Duration)
    )
  }

  trace_root <- "./pdp/results/star-fletcher-cpu-pdp-trace-tree"

  starvation_by_config <- list.files(trace_root, pattern = "^\\.phase1-done$", all.files = TRUE, recursive = TRUE) |>
    dirname() |>
    map(\(leaf) {
      parts <- str_split_1(leaf, "/")  # <nó>/OutputTime-<x>/BlockSeg-<n>
      starvation_stats(file.path(trace_root, leaf)) |>
        mutate(
          Machine = sub("\\d+$", "", parts[1]),
          OutputTime = as.numeric(str_remove(parts[2], "OutputTime-")),
          BlockSeg = as.integer(str_remove(parts[3], "BlockSeg-")),
          .before = 1
        )
    }) |>
    list_rbind() |>
    mutate(
      Order = 384 / BlockSeg,
      IO = case_when(
        OutputTime >= 1 ~ "Without IO",
        near(OutputTime, 1e-4) ~ "IO every step",
        TRUE ~ "IO every 5 steps"
      ),
      IO = factor(IO, levels = c("Without IO", "IO every step", "IO every 5 steps")),
      .after = BlockSeg
    ) |>

** Análise da Escalabilidade

Resultados do experimento de escalabilidade: o desempenho de cada programa conforme aumenta o
número de cores usados (primeiro os p-cores e depois os e-cores, um thread por core físico).
O último ponto das máquinas com SMT usa todos os threads (hyperthreading) e não está em escala no
eixo X, por isso a linha até ele é pontilhada.

#+BEGIN_SRC R :session *R* :async yes :results output silent
  # Lê os resultados de escalabilidade. Cada máquina tem o seu csv em ./pdp/scalability e as
  # pastas de resultado têm o nome do nó (cei1, draco6, ...), por isso o csv é escolhido pelo
  # nome da máquina sem o número. Uma máquina pode ter mais de um csv, como o grace, que tem
  # replicações a mais em star-fletcher-scal-grace-extra.csv: todos são lidos.
  load_scalability <- function(program, exp_folder, parse_fn) {
    list.dirs(glue("{base_dir}/{exp_folder}"), recursive = FALSE) |>
      basename() |>
      map(\(node) {
        csvs <- list.files("./pdp/scalability", full.names = TRUE,
                           pattern = glue("^{program}-scal-{sub('\\\\d+$', '', node)}(-.+)?\\.csv$"))
        map(csvs, \(f) read_csv(f, show_col_types = FALSE)) |>
          list_rbind() |>
          mutate(Machine = node)
      }) |>
      list_rbind() |>
      mutate(
        Steps = floor(as.numeric(OutputTime) / as.numeric(TimeStep)),
        Config = if (program == "star-fletcher") BlockSeg else Schedulers,
        FILE = glue("{base_dir}/{exp_folder}/{Machine}/stdout-{NumCPUs}-{Config}-{Steps}-{Blocks}.out")
      ) |>
      parse_files(FILE, parse_fn) |>
      select(-Config)
  }

  sf_scale <- load_scalability("star-fletcher", "star-fletcher-cpu-pdp-exp-scale", function(lines) {
    tibble(
      Msamples = lines |> str_subset("Msamples/s") |> str_extract("\\d+.\\d+") |> as.numeric(),
      ExecTime = lines |> str_subset("Computation Elapsed time") |> str_extract("\\d+.\\d+") |> as.numeric()
    )
  }) |>
    mutate(Program = "star-fletcher")

  fb_scale <- load_scalability("fletcher-base", "fletcher-base-pdp-exp-scale", function(lines) {
    tibble(
      Msamples = lines |> str_subset("MSamples/s") |> str_extract("\\d+.\\d+") |> as.numeric(),
      ExecTime = lines |> str_subset("Total execution time") |> str_extract("\\d+.\\d+") |> as.numeric()
    )
  }) |>
    mutate(Program = paste0("fletcher-base (", Schedulers, ")"))

  scale_stats <- bind_rows(sf_scale, fb_scale) |>
    mutate(
      Machine = sub("\\d+$", "", Machine),
      IO = factor(if_else(WithIO == 0, "Without IO", "IO every step"), levels = c("Without IO", "IO every step"))
    ) |>
    group_by(Machine, Program, IO, NumCPUs, SMT) |>
    summarize(
      n = n(),
      Msamples_mean = mean(Msamples),
      Msamples_error = (3 * sd(Msamples)) / sqrt(n),
      ExecTime_mean = mean(ExecTime),
      ExecTime_error = (3 * sd(ExecTime)) / sqrt(n),
      .groups = "drop"
    ) |>
    # O ponto com hyperthreading (SMT) não fica na escala do eixo X: ele é colocado um pouco
    # depois do último ponto com cores físicos, para não ficar longe demais
    group_by(Machine) |>
    mutate(
      MaxPhysical = max(NumCPUs[SMT == 0]),
      X = if_else(SMT == 1, MaxPhysical * 1.2, NumCPUs)
    ) |>
    ungroup()
#+END_SRC

  # para serem as mesmas em todas as máquinas.

  # (ver cpu-topology.org)
  pcore_count <- c(poti = 8, tupi = 8)

  scalability_plot <- function(stats, metric, y_label, title, log_y = FALSE, log_x = FALSE) {
    machine_plot <- function(machine, is_last) {
      d <- filter(stats, Machine == machine) |>
        mutate(
          y = .data[[paste0(metric, "_mean")]],
          err = .data[[paste0(metric, "_error")]],
          # na escala log2 o ponto com SMT precisa de um espaço maior para não colar no último ponto
          X = if_else(SMT == 1, MaxPhysical * (if (log_x) 1.6 else 1.2), NumCPUs)
        )
      ticks <- distinct(d, X, NumCPUs, SMT) |>
        mutate(Label = if_else(SMT == 1, paste0(NumCPUs, "\n(SMT)"), as.character(NumCPUs)))
      max_physical <- first(d$MaxPhysical)

      p <- ggplot(d, aes(x = X, y = y, color = Program)) +
        geom_line(data = filter(d, SMT == 0), linewidth = 0.8)
      if (any(d$SMT == 1)) {
        # o trecho até o ponto com SMT é pontilhado, já que a distância não está em escala
        p <- p + geom_line(data = filter(d, SMT == 1 | NumCPUs == max_physical), linetype = "dotted", linewidth = 0.8)
      }
      if (machine %in% names(pcore_count)) {
        # linha tracejada onde terminam os p-cores e começam a ser usados os e-cores
        p <- p +
          geom_vline(xintercept = pcore_count[[machine]], linetype = "dashed", color = "gray40") +
          annotate("text", x = pcore_count[[machine]], y = Inf, label = " E-cores →",
                   hjust = 0, vjust = 1.5, size = 3.5, color = "gray30")
      }
      
      p +
        geom_point(alpha = 0.8) +
        # a largura da barra de erro é em unidades do eixo, que na escala log2 são bem maiores
        geom_errorbar(aes(ymin = y - err, ymax = y + err), width = if (log_x) 0.15 else 0.5) +
        (if (log_y) scale_y_log10() else scale_y_continuous(limits = c(0, NA))) +
        scale_x_continuous(
          transform = if (log_x) "log2" else "identity",
          breaks = ticks$X,
          labels = ticks$Label
        ) +
        scale_color_manual(values = program_colors, limits = names(program_colors)) +
        facet_grid(
          (if(unique(d$IO) |> length() == 1) ~ Machine else IO ~ Machine),
          scales = "free_y") +
        labs(x = "Number of cores/threads", y = y_label, color = NULL) +
        theme_bw(base_size = 15) +
        theme(axis.text.x = element_text(size = 9)) +
        # o rótulo do IO vale para a linha inteira, então só aparece no último gráfico
        (if (is_last) NULL else theme(strip.text.y = element_blank(), strip.background.y = element_blank()))
    }

    # só a máquina com mais programas mostra a legenda, para o patchwork juntar uma só
    machines <- sort(unique(stats$Machine))
    with_legend <- stats |> distinct(Machine, Program) |> count(Machine) |> slice_max(n, n = 1, with_ties = FALSE) |> pull(Machine)

    machines |>
      map(\(m) machine_plot(m, m == last(machines)) + (if (m == with_legend) NULL else guides(color = "none"))) |>
      wrap_plots(nrow = 1) +
      plot_layout(guides = "collect", axis_titles = "collect") +
      plot_annotation(
        title = title,
        subtitle = "Faceted by machine (columns) and IO (rows). The SMT point (all hardware threads) is not to scale."
      ) &
      theme(legend.position = "bottom")
  }

  scale_stats |>
    filter(IO == "Without IO") |>
    scalability_plot("Msamples", "Msamples/s", "MSamples/s vs. Number of Cores (without IO)", log_x = FALSE)
